local function AntiFlingLoop(name, getFolderFunc)
    local active = false
    task.spawn(function()
        while true do
            if active and LocalPlayer.Character then
                local folder = getFolderFunc()
                if folder then
                    for _, item in ipairs(folder:GetChildren()) do
                        local isMine = false
                        if name == "Vehicles" then
                            for _, seat in ipairs(item:GetDescendants()) do
                                if (seat:IsA("VehicleSeat") or seat:IsA("Seat")) and seat.Occupant and seat.Occupant.Parent == LocalPlayer.Character then
                                    isMine = true
                                    break
                                end
                            end
                        elseif name == "Canoes" then
                            local owner = item:FindFirstChild("Owner")
                            isMine = owner and owner.Value == LocalPlayer
                        elseif name == "Jets" or name == "Helis" then
                            isMine = item.Name == LocalPlayer.Name
                        end
                        if not isMine then
                            table.insert(backupTables[name], item:Clone())
                            item:Destroy()
                        end
                    end
                end
            end
            task.wait(0.03)
        end
    end)
    return function(state)
        active = state
        TeleportCarro:MostrarNotificacao(name.." "..(state and "ativado!" or "desativado!"))
        if not state then
            for _, item in ipairs(backupTables[name]) do
                local parentFolder = getFolderFunc()
                if parentFolder then item.Parent = parentFolder end
            end
            backupTables[name] = {}
        end
    end
end

TabAnti:AddToggle({
    Name = "مضاد للتأخير",
    Description = "",
    Default = false,
    Callback = function(state)
        local Players = game:GetService("Players")
        local dedupLock = {}
        local IGNORED_PLAYER

        if not state then return end

        local function marcarIgnorado(player)
            IGNORED_PLAYER = player
        end

        local function isTargetTool(inst)
            return inst:IsA("Tool")
        end

        local function gatherTools(player)
            local found = {}
            local containers = {}
            if player.Character then table.insert(containers, player.Character) end
            local backpack = player:FindFirstChildOfClass("Backpack")
            if backpack then table.insert(containers, backpack) end
            local sg = player:FindFirstChild("StarterGear")
            if sg then table.insert(containers, sg) end
            for _, container in ipairs(containers) do
                for _, child in ipairs(container:GetChildren()) do
                    if isTargetTool(child) then table.insert(found, child) end
                end
            end
            return found
        end

        local function dedupePlayer(player)
            if player == IGNORED_PLAYER then return end
            if dedupLock[player] then return end
            dedupLock[player] = true
            local tools = gatherTools(player)
            if #tools > 1 then
                for i = 2, #tools do pcall(function() tools[i]:Destroy() end) end
            end
            dedupLock[player] = false
        end

        local function hookPlayer(player)
            if not IGNORED_PLAYER then marcarIgnorado(player) end
            task.defer(dedupePlayer, player)
            local function setupChar(char)
                task.delay(0.5, function() dedupePlayer(player) end)
                char.ChildAdded:Connect(function(child)
                    if isTargetTool(child) then task.delay(0.1, function() dedupePlayer(player) end) end
                end)
            end
            if player.Character then setupChar(player.Character) end
            player.CharacterAdded:Connect(setupChar)
            local backpack = player:WaitForChild("Backpack", 10)
            if backpack then
                backpack.ChildAdded:Connect(function(child)
                    if isTargetTool(child) then task.delay(0.1, function() dedupePlayer(player) end) end
                end)
            end
            local sg = player:FindFirstChild("StarterGear") or player:WaitForChild("StarterGear", 10)
            if sg then
                sg.ChildAdded:Connect(function(child)
                    if isTargetTool(child) then task.delay(0.1, function() dedupePlayer(player) end) end
                end)
            end
        end

        Players.PlayerAdded:Connect(hookPlayer)
        for _, plr in ipairs(Players:GetPlayers()) do hookPlayer(plr) end

        task.spawn(function()
            while state do
                for _, plr in ipairs(Players:GetPlayers()) do dedupePlayer(plr) end
                task.wait(2)
            end
        end)
    end
})

-- محرك السرعة القصوى (تجاوز المضاد بنسبة 100%)
function createGodSpeedBang(name, config)
    local active = false
    local connection = nil
    local togglePos = false

    BangTab:AddToggle({Name = name, Default = false, Callback = function(Value)
        active = Value
        local char = LocalPlayer.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        
        if Value then
            hum.PlatformStand = true
            -- استخدام PreRender هو السر لسبق أي سكربت مضاد في السيرفر
            connection = RunService.PreRender:Connect(function()
                if active and getgenv().selectedPlayer then
                    local target = Players:FindFirstChild(getgenv().selectedPlayer)
                    if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                        local targetRoot = target.Character.HumanoidRootPart
                        
                        -- Bypass Logic: تجاهل المواقع الوهمية (تحت الخريطة)
                        local targetCF = targetRoot.CFrame
                        if targetCF.Y < -1000 then 
                            -- محاولة استعادة الموقع الأصلي إذا كان الخصم يحاول الهرب تحت الأرض
                            targetCF = CFrame.new(targetCF.X, 0, targetCF.Z) 
                        end

                        -- سرعة الاهتزاز (الترليونية)
                        local dist = (togglePos and config.Dist1) or config.Dist2
                        local finalCF
                        
                        if config.Type == "Back" then
                            finalCF = targetCF * CFrame.new(0, config.Y, dist)
                        elseif config.Type == "Face" then
                            finalCF = targetCF * CFrame.new(0, config.Y, -dist) * CFrame.Angles(0, math.rad(180), 0)
                        elseif config.Type == "Hand" then
                            finalCF = targetCF * CFrame.new(0.6, config.Y, -dist) * CFrame.Angles(0, math.rad(180), math.rad(60))
                        end
                        
                        root.CFrame = finalCF
                        -- تغيير الحالة في كل فريم برمجياً (أسرع من أي انتظار)
                        togglePos = not togglePos
                    end
                end
            end)
        else
            if connection then connection:Disconnect() end
            hum.PlatformStand = false
        end
    end})
end

function createUltraBang(name, config)
    local active = false
    local connection = nil
    local togglePos = false

    BangTab:AddToggle({Name = name, Default = false, Callback = function(Value)
        active = Value
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        
        if Value then
            hum.PlatformStand = true
            -- استخدام RenderStepped لسبق المضاد في كل فريم
            connection = RunService.RenderStepped:Connect(function()
                if active and getgenv().selectedPlayer then
                    local target = Players:FindFirstChild(getgenv().selectedPlayer)
                    if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                        local targetRoot = target.Character.HumanoidRootPart
                        
                        -- نظام الـ Bypass الحقيقي: إذا كان مكانه تحت -100000 (يعني مفعل مضاد) يتجاهل الإحداثية
                        local targetCF = targetRoot.CFrame
                        if targetCF.Y < -50000 then return end 

                        local speed = config.Speed or 0.05
                        local dist = (togglePos and config.Dist1) or config.Dist2
                        
                        local finalCF
                        if config.Type == "Back" then
                            finalCF = targetCF * CFrame.new(0, config.Y, dist)
                        elseif config.Type == "Face" then
                            finalCF = targetCF * CFrame.new(0, config.Y, -dist) * CFrame.Angles(0, math.rad(180), 0)
                        elseif config.Type == "Hand" then -- حركة اليد (التجليخ)
                            finalCF = targetCF * CFrame.new(0.5, config.Y, -dist) * CFrame.Angles(0, math.rad(180), math.rad(45))
                        end
                        
                        root.CFrame = finalCF
                        togglePos = not togglePos
                    end
                end
            end)
        else
            if connection then connection:Disconnect() end
            hum.PlatformStand = false
        end
    end})
end

eTab:AddButton({
    Name = "تنفيذ القتل",
    Callback = function()
        if not selectedPlayerName or not Players:FindFirstChild(selectedPlayerName) then return end
        local target = Players:FindFirstChild(selectedPlayerName)
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then return end

        local originalPos = root.CFrame

        -- وظيفة جلب المركبة وتعزيز القوة
        local function GetVehicle()
            local v = workspace:FindFirstChild("Vehicles")
            return v and (v:FindFirstChild(LocalPlayer.Name .. "Car") or v:FindFirstChild(LocalPlayer.Name .. "Couch"))
        end

        local veh = GetVehicle()

        -- إذا لم تكن هناك مركبة، استدعي الباص فوراً (سرعة البرق)
        if not veh then
            root.CFrame = CFrame.new(1118.81, 75.998, -1138.61) -- إحداثيات ريسباون السيارات
            task.wait(0.1)
            local re = ReplicatedStorage:FindFirstChild("RE")
            if re and re:FindFirstChild("1Ca1r") then
                re["1Ca1r"]:FireServer("PickingCar", "SchoolBus")
            end
            repeat task.wait() veh = GetVehicle() until veh or not task.wait(1)
        end

        if veh then
            -- ركوب تلقائي فوري
            local seat = veh:FindFirstChildWhichIsA("VehicleSeat", true)
            if seat and not char.Humanoid.Sit then
                root.CFrame = seat.CFrame
                repeat task.wait() until char.Humanoid.Sit
            end

            -- خوارزمية الذبذبات المدمرة (أقوى دوران بالعالم)
            local killLoop
            killLoop = RunService.RenderStepped:Connect(function()
                if not target.Character or not veh.Parent or not char.Humanoid.Sit then
                    killLoop:Disconnect()
                    return
                end

                local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
                local tHum = target.Character:FindFirstChildOfClass("Humanoid")

                if tHum and tHum.Sit then
                    -- القتل النهائي (Void Banisher)
                    veh:SetPrimaryPartCFrame(CFrame.new(0, -500000, 0))
                    task.wait(0.1)
                    char.Humanoid.Jump = true
                    root.CFrame = originalPos
                    killLoop:Disconnect()
                else
                    -- ذبذبات جنونية (تردد 150 هرتز)
                    local t = tick() * 150
                    local rotOffset = CFrame.Angles(math.rad(t), math.rad(t), math.rad(t))
                    local posOffset = Vector3.new(math.sin(t)*15, math.cos(t)*5, math.cos(t)*15)
                    
                    -- تطبيق القوة الفيزيائية الخارقة
                    veh:SetPrimaryPartCFrame(CFrame.new(tRoot.Position + posOffset) * rotOffset)
                    
                    -- قوة دفع جانبية لمنع الهروب
                    veh.PrimaryPart.Velocity = Vector3.new(1000, 1000, 1000)
                    veh.PrimaryPart.RotVelocity = Vector3.new(1000, 1000, 1000)
                end
            end)
        end
    end
})
